$$ \text{好瓜} \leftrightarrow \big((\text{色泽}=*)\wedge(\text{根蒂}=\text{蜷缩})\wedge(\text{敲声}=*)\big) \vee \big((\text{色泽}=\text{乌黑})\wedge(\text{根蒂}=*)\wedge(\text{敲声}=\text{沉闷})\big), $$
